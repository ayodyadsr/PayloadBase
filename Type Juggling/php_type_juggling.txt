# ---- Type Juggling with == ----
# PHP: "0" == false → true
# "0" == null → false
# "" == false → true
# "" == null → true
# "php" == 0 → true (before PHP 8)
# "1abc" == 1 → true (before PHP 8)
0 == "foo"        # true (PHP 7)
0 == ""           # true (PHP 7)
0 == "0"          # true
0 == false        # true
0 == null         # true
"" == null        # true
"" == false       # true
"1" == "01"       # true
"10" == "1e1"     # true
100 == "1e2"      # true
"0" == false      # true
"0" == 0          # true
"0" == null       # false (!)
