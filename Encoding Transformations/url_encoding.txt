# ============================================================
# Encoding Transformations — URL Encoding
# ============================================================

# ---- URL Encode Special Characters ----
%20  = space
%21  = !
%22  = "
%23  = #
%24  = $
%25  = %
%26  = &
%27  = '
%28  = (
%29  = )
%2a  = *
%2b  = +
%2c  = ,
%2f  = /
%3a  = :
%3b  = ;
%3c  = <
%3d  = =
%3e  = >
%3f  = ?
%40  = @
%5b  = [
%5c  = \
%5d  = ]
%5e  = ^
%60  = `
%7b  = {
%7c  = |
%7d  = }
%7e  = ~
%0a  = LF (newline)
%0d  = CR (carriage return)
%00  = NULL
%09  = TAB

# ---- Double URL Encoding ----
%2520 = %20 = space
%252f = %2f = /
%255c = %5c = \
%2527 = %27 = '
%2522 = %22 = "
%2528 = %28 = (
%2529 = %29 = )
%2500 = %00 = NULL

# ---- Unicode/UTF-8 Overlong Encoding ----
%c0%af = / (overlong UTF-8)
%c0%ae = . (overlong UTF-8)
%c1%9c = \ (overlong UTF-8)
%e0%80%af = / (3-byte overlong)
%f0%80%80%af = / (4-byte overlong)
